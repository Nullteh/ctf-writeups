"""  
decrypt\_firmware\_fixed.py

Try AES-CTR decrypt variants using a provided key string.

\- Accepts either a Base64-encoded key (recommended) OR a raw ASCII key string.  
\- Builds nonce \= first 12 bytes of the decoded key (if available).  
\- Tries counter start values 0 and 1, and both big/little endian 4-byte counters.  
\- Writes output files for each variant and prints simple heuristics.

Usage:  
  python3 decrypt\_firmware\_fixed.py \--in firmware.enc \--key tCncHUA7iMUCHqQ9bMGTjg==

Requires: pip3 install pycryptodome  
"""

import argparse  
import base64  
import os  
import sys  
from Cryptodome.Cipher import AES  
from Cryptodome.Util import Counter

def maybe\_base64\_decode(s):  
    """If s looks like base64 (contains '+' '/' '=' or only base64 chars) try to decode."""  
    try:  
        \# Heuristic: if contains '=' or all chars are in base64 alphabet, try decode  
        b \= base64.b64decode(s, validate=True)  
        return b  
    except Exception:  
        return None

def printable\_ratio(b):  
    if not b:  
        return 0.0  
    printable \= sum(1 for x in b if 32 \<= x \<= 126 or x in (9,10,13))  
    return printable / len(b)

def try\_variant\_write(key\_bytes, nonce12, start, endian, ct\_bytes, out\_prefix):  
    \# build 4-byte counter appended to nonce12  
    if endian \== "big":  
        cnt \= (start).to\_bytes(4, "big")  
    else:  
        cnt \= (start).to\_bytes(4, "little")  
    iv \= nonce12 \+ cnt  
    iv\_int \= int.from\_bytes(iv, "big")   \# initial\_value taken as big-endian integer of full 16B block  
    ctr \= Counter.new(128, initial\_value=iv\_int)  
    cipher \= AES.new(key\_bytes, AES.MODE\_CTR, counter=ctr)  
    pt \= cipher.decrypt(ct\_bytes)  
    fname \= f"{out\_prefix}\_start{start}\_{endian}.bin"  
    with open(fname, "wb") as f:  
        f.write(pt)  
    return fname, len(pt), printable\_ratio(pt), pt\[:16\].hex()

def main():  
    p \= argparse.ArgumentParser()  
    p.add\_argument("--in", dest="infile", required=True)  
    p.add\_argument("--key", dest="key\_str", required=True,  
                   help="Base64 key (recommended) or raw ASCII key string")  
    p.add\_argument("--out", dest="out\_prefix", default="firmware.dec")  
    args \= p.parse\_args()

    if not os.path.exists(args.infile):  
        print("\[\!\] Input not found:", args.infile); sys.exit(1)

    \# 1\) load ciphertext  
    ct \= open(args.infile, "rb").read()  
    print(f"\[\*\] Read ciphertext {args.infile} ({len(ct)} bytes)")

    \# 2\) try to decode key from base64, else use raw ascii bytes  
    key\_bytes \= maybe\_base64\_decode(args.key\_str)  
    if key\_bytes is None:  
        print("\[\*\] Key does not decode as base64, using raw ASCII bytes of provided string.")  
        key\_bytes \= args.key\_str.encode("utf-8")  
    else:  
        print("\[\*\] Key decoded from base64 ({} bytes).".format(len(key\_bytes)))

    print("\[\*\] Key length (bytes):", len(key\_bytes))  
    if len(key\_bytes) not in (16, 24, 32):  
        print("\[\!\] Warning: key length is not a standard AES size (16/24/32). You may get an error or wrong output.")

    \# 3\) determine nonce12  
    if len(key\_bytes) \>= 12:  
        nonce12 \= key\_bytes\[:12\]  
        print("\[\*\] Using nonce \= first 12 bytes of key (hex):", nonce12.hex())  
    else:  
        print("\[\!\] Key \< 12 bytes, cannot extract 12-byte nonce from key. Abort.")  
        sys.exit(1)

    \# 4\) try variants  
    results \= \[\]  
    for start in (0,1):  
        for endian in ("big","little"):  
            try:  
                fname, size, pratio, magic16 \= try\_variant\_write(key\_bytes, nonce12, start, endian, ct, args.out\_prefix)  
                print(f"\[+\] Wrote {fname} size={size} printable\_ratio={pratio:.3f} magic16={magic16}")  
                results.append((fname, pratio, magic16))  
            except ValueError as e:  
                print(f"\[\!\] Error for start={start}, endian={endian}: {e}")

    \# 5\) heuristic pick  
    best \= sorted(results, key=lambda x: x\[1\], reverse=True)  
    if best:  
        print("\\n\[\*\] Best candidate by printable ratio (top 3):")  
        for fname, pratio, magic16 in best\[:3\]:  
            print(f"  {fname}: printable\_ratio={pratio:.3f}, magic16={magic16}")  
    else:  
        print("\[\!\] No outputs produced.")

if \_\_name\_\_ \== "\_\_main\_\_":  
    main()
